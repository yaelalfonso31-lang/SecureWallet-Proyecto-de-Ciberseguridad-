# PolÃ­tica de seguridad del equipo

## Reglas obligatorias
1. **Nunca** subir secretos: `.env`, llaves privadas, contraseÃ±as, tokens. (GitHub Push Protection y Gitleaks lo vigilan.)
2. **Solo datos de prueba.** Prohibido usar tarjetas, cuentas o datos personales reales.
3. **Nunca** guardar CVV. El nÃºmero de tarjeta (PAN) solo se almacena cifrado (AES-256-GCM); fuera de la Wallet circula el *token*.
4. **No registrar** (logs, consola, bitÃ¡cora) contraseÃ±as, PAN, tokens ni llaves.
5. ContraseÃ±as con **Argon2id**; nada de MD5/SHA simple para contraseÃ±as.
6. Toda consulta SQL va parametrizada (usar SQLAlchemy). Toda entrada se valida con Pydantic.
7. Cambios en `backend/shared/` y `mobile/lib/core/` requieren revisiÃ³n del arquitecto de seguridad.
8. Dependencias nuevas: revisar que estÃ©n mantenidas y sin vulnerabilidades conocidas.

## Reportar un problema
Abre un Issue con la etiqueta `security` (repositorio privado) o avisa directamente al arquitecto de seguridad. Si un secreto se subiÃ³ por error, **rÃ³talo de inmediato** (cambiar el valor), no basta con borrar el commit.

"""AutenticaciÃ³n y autorizaciÃ³n del Banco.

El Banco emite sus propios JWT, firmados con `BANK_JWT_SECRET` y con `aud="bank"`,
independientes de los tokens de la Wallet. Estas dependencias son el contrato que
comparten las dos personas del equipo Banco (definido en B0).
"""
from datetime import datetime, timedelta, timezone

import jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from sqlalchemy.orm import Session

from bank_service.database import get_db
from bank_service.models import BankCustomer
from shared.config import settings

bearer = HTTPBearer(auto_error=False)

BANK_TOKEN_AUDIENCE = "bank"

# Scopes que puede tener un token del Banco. El fixture de pruebas los concede todos.
ALL_SCOPES = (
    "accounts:read",
    "promotions:read",
    "promotions:redeem",
    "transfers:write",
)


def create_bank_token(customer_id: str, scopes=ALL_SCOPES) -> str:
    """Emite un JWT del Banco para un cliente, con `aud="bank"` y los scopes dados."""
    if not settings.bank_jwt_secret:
        raise RuntimeError("BANK_JWT_SECRET no estÃ¡ configurada (ver scripts/generate_env.py)")
    now = datetime.now(timezone.utc)
    payload = {
        "sub": customer_id,
        "aud": BANK_TOKEN_AUDIENCE,
        "scope": " ".join(scopes),
        "iat": now,
        "exp": now + timedelta(minutes=settings.access_token_minutes),
    }
    return jwt.encode(payload, settings.bank_jwt_secret, algorithm=settings.jwt_algorithm)


def _decode_bank_token(token: str) -> dict:
    """Valida firma, expiraciÃ³n y `aud="bank"`. Lanza jwt.PyJWTError si falla."""
    return jwt.decode(
        token,
        settings.bank_jwt_secret,
        algorithms=[settings.jwt_algorithm],
        audience=BANK_TOKEN_AUDIENCE,
    )


def _get_payload(creds: HTTPAuthorizationCredentials | None) -> dict:
    unauthorized = HTTPException(status.HTTP_401_UNAUTHORIZED, "No autenticado")
    if creds is None:
        raise unauthorized
    try:
        return _decode_bank_token(creds.credentials)
    except jwt.PyJWTError:
        raise unauthorized


def _customer_from_payload(payload: dict, db: Session) -> BankCustomer:
    customer = db.get(BankCustomer, payload.get("sub"))
    if customer is None:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "No autenticado")
    return customer


def get_current_customer(
    creds: HTTPAuthorizationCredentials | None = Depends(bearer),
    db: Session = Depends(get_db),
) -> BankCustomer:
    """Devuelve el cliente dueÃ±o del token (cargado desde la BD)."""
    return _customer_from_payload(_get_payload(creds), db)


def require_scope(scope: str):
    """FÃ¡brica de dependencias: exige que el token incluya `scope`."""

    def dependency(
        creds: HTTPAuthorizationCredentials | None = Depends(bearer),
        db: Session = Depends(get_db),
    ) -> BankCustomer:
        payload = _get_payload(creds)
        if scope not in payload.get("scope", "").split():
            raise HTTPException(status.HTTP_403_FORBIDDEN, f"Scope requerido: {scope}")
        return _customer_from_payload(payload, db)

    return dependency


def require_admin(customer: BankCustomer = Depends(get_current_customer)) -> BankCustomer:
    """Exige rol admin. El rol se lee de la BD (vÃ­a get_current_customer), no del token."""
    if customer.role != "admin":
        raise HTTPException(status.HTTP_403_FORBIDDEN, "Se requiere rol de administrador")
    return customer

import json
import time
import uuid

from fastapi import APIRouter, Depends, Header, HTTPException
from sqlalchemy import or_, select
from sqlalchemy.orm import Session

from bank_service.database import get_db
from bank_service.models import Account, ProcessedOrder
from shared.config import settings
from shared.schemas import SignedOrder, SignedReceipt
from shared.security.hashing import constant_time_equals, sha256_hex
from shared.security.signing import canonical_json, sign, verify
from shared.timeutil import utcnow

router = APIRouter(tags=["transactions"])

DEMO_INITIAL_BALANCE_CENTS = 100_000  # $1,000.00 MXN de saldo de prueba


def require_api_key(x_api_key: str | None = Header(default=None)) -> None:
    """AutenticaciÃ³n servicio-a-servicio (Tienda -> Banco).

    TODO equipo Banco: reemplazar por mTLS entre servicios.
    """
    expected = settings.bank_api_key
    if not expected or not x_api_key or not constant_time_equals(x_api_key, expected):
        raise HTTPException(401, "API key invÃ¡lida")


def _get_or_create_account(db: Session, owner_ref: str) -> Account:
    account = db.scalar(select(Account).where(Account.owner_ref == owner_ref))
    if account is None:
        account = Account(owner_ref=owner_ref, balance_cents=DEMO_INITIAL_BALANCE_CENTS)
        db.add(account)
        db.flush()
    return account


@router.get("/bank/public-key")
def bank_public_key():
    """Clave pÃºblica con la que se verifican los recibos firmados por el Banco."""
    return {"algorithm": "Ed25519", "public_key": settings.bank_signing_public_key}


@router.post("/transactions/process", response_model=SignedReceipt, dependencies=[Depends(require_api_key)])
def process_transaction(body: SignedOrder, db: Session = Depends(get_db)):
    order = body.order.model_dump()

    # 1) Autenticidad e integridad: la firma debe ser de la Wallet y la orden no pudo cambiar.
    if not verify(settings.wallet_signing_public_key, order, body.signature):
        raise HTTPException(400, "Firma invÃ¡lida")

    # 2) Frescura: rechaza Ã³rdenes viejas (ventana corta anti-replay).
    if abs(time.time() - order["timestamp"]) > settings.order_max_age_seconds:
        raise HTTPException(400, "Orden expirada")

    # 3) Unicidad: el mismo nonce/order_id no se procesa dos veces.
    duplicate = db.scalar(
        select(ProcessedOrder).where(
            or_(ProcessedOrder.nonce == order["nonce"], ProcessedOrder.order_id == order["order_id"])
        )
    )
    if duplicate:
        raise HTTPException(409, "Orden ya procesada")

    # 4) Fondos.
    account = _get_or_create_account(db, order["payer"])
    if account.balance_cents < order["amount_cents"]:
        raise HTTPException(402, "Fondos insuficientes")
    account.balance_cents -= order["amount_cents"]

    # 5) Recibo firmado por el Banco (no repudio).
    receipt = {
        "receipt_id": uuid.uuid4().hex,
        "order_id": order["order_id"],
        "payer": order["payer"],
        "amount_cents": order["amount_cents"],
        "currency": order["currency"],
        "status": "approved",
        "processed_at": utcnow().isoformat(),
        "order_hash": sha256_hex(canonical_json(order)),
    }
    signature = sign(settings.bank_signing_private_key, receipt)

    db.add(
        ProcessedOrder(
            order_id=order["order_id"],
            nonce=order["nonce"],
            payer=order["payer"],
            amount_cents=order["amount_cents"],
            receipt_json=json.dumps(receipt),
        )
    )
    db.commit()
    return SignedReceipt(receipt=receipt, signature=signature)

"""Genera un archivo .env con secretos aleatorios para desarrollo.

Uso (desde la carpeta backend/):

    python scripts/generate_env.py --force  # sobreescribe

import base64
import os
import secrets
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from shared.security.signing import generate_keypair  # noqa: E402

TARGET = Path(__file__).resolve().parents[1] / ".env"


def main() -> None:
    if TARGET.exists() and "--force" not in sys.argv:
        print(f"{TARGET} ya existe. Usa --force para sobreescribirlo.")
        return

    wallet_priv, wallet_pub = generate_keypair()
    bank_priv, bank_pub = generate_keypair()
    data_key = base64.urlsafe_b64encode(os.urandom(32)).decode()

    content = f"""ENVIRONMENT=dev

WALLET_DATABASE_URL=sqlite:///./wallet.db
BANK_DATABASE_URL=sqlite:///./bank.db
STORE_DATABASE_URL=sqlite:///./store.db

JWT_SECRET={secrets.token_urlsafe(48)}
ACCESS_TOKEN_MINUTES=15
BANK_JWT_SECRET={secrets.token_urlsafe(48)}

DATA_ENCRYPTION_KEY={data_key}

WALLET_SIGNING_PRIVATE_KEY={wallet_priv}

def make_session_factory(url: str):
    """Crea (engine, SessionLocal) para una URL de base de datos."""
    kwargs = {}
    if url.startswith("sqlite"):
        kwargs["connect_args"] = {"check_same_thread": False}
    engine = create_engine(url, **kwargs)
    session_local = sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)
    return engine, session_local

"""Cifrado autenticado AES-256-GCM para datos en reposo (p. ej. nÃºmero de tarjeta)."""
import base64
import os

from cryptography.hazmat.primitives.ciphers.aead import AESGCM

from shared.config import settings


def generate_key() -> str:
    return base64.urlsafe_b64encode(AESGCM.generate_key(bit_length=256)).decode()


def _load_key() -> bytes:
    if not settings.data_encryption_key:
        raise RuntimeError("DATA_ENCRYPTION_KEY no estÃ¡ configurada (ver scripts/generate_env.py)")
    key = base64.urlsafe_b64decode(settings.data_encryption_key)
    if len(key) != 32:
        raise RuntimeError("DATA_ENCRYPTION_KEY debe ser de 32 bytes")
    return key


def encrypt(plaintext: str, aad: str = "") -> str:
    """Cifra y devuelve base64url(nonce || ciphertext+tag).

    `aad` (datos asociados, p. ej. el id del usuario) liga el cifrado a su dueÃ±o:
    si alguien copia el valor a otro registro, el descifrado falla.
    """
    nonce = os.urandom(12)  # nonce Ãºnico por cifrado, nunca reutilizar
    ciphertext = AESGCM(_load_key()).encrypt(nonce, plaintext.encode(), aad.encode() or None)
    return base64.urlsafe_b64encode(nonce + ciphertext).decode()


def decrypt(token: str, aad: str = "") -> str:
    raw = base64.urlsafe_b64decode(token)
    nonce, ciphertext = raw[:12], raw[12:]
    return AESGCM(_load_key()).decrypt(nonce, ciphertext, aad.encode() or None).decode()

"""Firma electrÃ³nica con Ed25519 sobre JSON canÃ³nico."""
import base64
import json

from cryptography.exceptions import InvalidSignature
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.ed25519 import (
    Ed25519PrivateKey,
    Ed25519PublicKey,
)


def _b64(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).decode()


def generate_keypair() -> tuple[str, str]:
    """Devuelve (clave_privada, clave_pÃºblica) en base64url."""
    private = Ed25519PrivateKey.generate()
    private_bytes = private.private_bytes(
        serialization.Encoding.Raw,
        serialization.PrivateFormat.Raw,
        serialization.NoEncryption(),
    )
    public_bytes = private.public_key().public_bytes(
        serialization.Encoding.Raw, serialization.PublicFormat.Raw
    )
    return _b64(private_bytes), _b64(public_bytes)


def canonical_json(obj) -> bytes:
    """SerializaciÃ³n determinista: ambas partes deben firmar/verificar los mismos bytes.

    La app Flutter implementa la misma regla (ver mobile/lib/core/security/receipt_verifier.dart).
    """
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode()


def sign(private_key_b64: str, payload: dict) -> str:
    if not private_key_b64:
        raise RuntimeError("Falta la clave privada de firma (ver scripts/generate_env.py)")
    key = Ed25519PrivateKey.from_private_bytes(base64.urlsafe_b64decode(private_key_b64))
    return _b64(key.sign(canonical_json(payload)))


def verify(public_key_b64: str, payload: dict, signature_b64: str) -> bool:
    if not public_key_b64:
        raise RuntimeError("Falta la clave pÃºblica de verificaciÃ³n")
    try:
        key = Ed25519PublicKey.from_public_bytes(base64.urlsafe_b64decode(public_key_b64))
        key.verify(base64.urlsafe_b64decode(signature_b64), canonical_json(payload))
        return True
    except (InvalidSignature, ValueError):
        return False

def luhn_valid(number: str) -> bool:
    """Algoritmo de Luhn para validar nÃºmeros de tarjeta."""
    if not number.isdigit():
        return False
    total = 0
    for i, ch in enumerate(reversed(number)):
        digit = int(ch)
        if i % 2 == 1:
            digit *= 2
            if digit > 9:
                digit -= 9
        total += digit
    return total % 10 == 0


def detect_brand(number: str) -> str:
    if number.startswith("4"):
        return "visa"
    if number[:2] in {"51", "52", "53", "54", "55"} or (
        number[:4].isdigit() and 2221 <= int(number[:4]) <= 2720
    ):
        return "mastercard"
    if number[:2] in {"34", "37"}:
        return "amex"
    return "desconocida"

@router.post("/{order_id}/pay")
def pay_order(order_id: str, body: SignedOrder, db: Session = Depends(get_db)):
    """Recibe la orden firmada por la Wallet, la manda al Banco y verifica el recibo firmado."""
    order = db.get(Order, order_id)
    if order is None:
        raise HTTPException(404, "Orden no encontrada")
    if order.status != "pending":
        raise HTTPException(409, "La orden ya fue pagada")

    signed = body.order
    if (
        signed.order_id != order_id
        or signed.amount_cents != order.total_cents
        or signed.merchant != MERCHANT
        or signed.currency != order.currency
    ):
        raise HTTPException(400, "La orden firmada no coincide con la compra")

    try:
        response = process_payment(body.model_dump())
    except httpx.HTTPError:
        raise HTTPException(502, "No fue posible contactar al banco")

    if response.status_code != 200:
        try:
            detail = response.json().get("detail", "Pago rechazado")
        except ValueError:
            detail = "Pago rechazado"
        code = response.status_code if response.status_code in (400, 402, 409) else 502
        raise HTTPException(code, detail)

    data = response.json()
    receipt, signature = data["receipt"], data["signature"]

    # El recibo debe venir firmado por el Banco y coincidir con la compra.
    if not verify(settings.bank_signing_public_key, receipt, signature):
        raise HTTPException(502, "Recibo del banco con firma invÃ¡lida")
    if (
        receipt.get("status") != "approved"
        or receipt.get("order_id") != order_id
        or receipt.get("amount_cents") != order.total_cents
    ):
        raise HTTPException(502, "Recibo del banco inconsistente")

    for item in json.loads(order.items_json):
        product = db.get(Product, item["product_id"])
        product.stock -= item["quantity"]
    order.status = "paid"
    order.receipt_json = json.dumps(receipt)
    order.receipt_signature = signature
    db.commit()

    return {"order_id": order_id, "status": "paid", "receipt": receipt, "signature": signature}

from sqlalchemy import select
from sqlalchemy.orm import Session

from shared.security.hashing import chain_hash
from shared.timeutil import utcnow
from wallet_service.models import AuditLog

GENESIS = "0" * 64


def _record(event: str, user_id, detail: str, created_at) -> dict:
    return {
        "event": event,
        "user_id": user_id,
        "detail": detail,
        "created_at": created_at.isoformat(),
    }


def log_event(db: Session, event: str, user_id: str | None = None, detail: str = "") -> None:
    """Registra un evento. Nunca guardes contraseÃ±as, PAN ni tokens en `detail`."""
    last = db.scalars(select(AuditLog).order_by(AuditLog.id.desc())).first()
    prev_hash = last.hash if last else GENESIS
    created_at = utcnow()
    entry = AuditLog(
        event=event,
        user_id=user_id,
        detail=detail,
        prev_hash=prev_hash,
        hash=chain_hash(prev_hash, _record(event, user_id, detail, created_at)),
        created_at=created_at,
    )
    db.add(entry)
    db.commit()


def verify_chain(db: Session) -> bool:
    """True si ningÃºn registro de la bitÃ¡cora fue alterado."""
    prev_hash = GENESIS
    for row in db.scalars(select(AuditLog).order_by(AuditLog.id)):
        expected = chain_hash(prev_hash, _record(row.event, row.user_id, row.detail, row.created_at))
        if row.prev_hash != prev_hash or row.hash != expected:
            return False
        prev_hash = row.hash
    return True

# Hash ficticio para igualar tiempos de respuesta cuando el correo no existe.
_DUMMY_HASH = hash_password("contraseÃ±a-ficticia-1")
_INVALID = "Credenciales invÃ¡lidas"


@router.post("/register", response_model=UserOut, status_code=201)
def register(body: RegisterIn, db: Session = Depends(get_db)):
    email = body.email.lower()
    if db.scalar(select(User).where(User.email == email)):
        # Mensaje genÃ©rico para no revelar quÃ© correos existen.
        raise HTTPException(409, "No fue posible completar el registro")
    user = User(
        email=email,
        full_name=body.full_name.strip(),
        password_hash=hash_password(body.password),
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    log_event(db, "register", user.id)
    return user


@router.post("/login", response_model=TokenOut)
def login(body: LoginIn, db: Session = Depends(get_db)):
    user = db.scalar(select(User).where(User.email == body.email.lower()))

    if user is None:
        verify_password(_DUMMY_HASH, body.password)
        raise HTTPException(401, _INVALID)

    if user.locked_until and user.locked_until > utcnow():
        log_event(db, "login_blocked", user.id)
        raise HTTPException(423, "Cuenta bloqueada temporalmente. Intenta mÃ¡s tarde.")

    if not verify_password(user.password_hash, body.password):
        user.failed_attempts += 1
        if user.failed_attempts >= settings.max_failed_logins:
            user.locked_until = utcnow() + timedelta(minutes=settings.lockout_minutes)
            user.failed_attempts = 0
            log_event(db, "account_locked", user.id)
        db.commit()
        log_event(db, "login_failed", user.id)
        raise HTTPException(401, _INVALID)

    user.failed_attempts = 0
    user.locked_until = None
    db.commit()
    log_event(db, "login_ok", user.id)
    return TokenOut(access_token=create_access_token(user.id))

    number = re.sub(r"[ -]", "", body.number)
    if not (number.isdigit() and 13 <= len(number) <= 19 and luhn_valid(number)):

    )
    db.add(card)
    db.commit()

    log_event(db, "card_added", user.id, detail=f"last4={card.last4}")

    """La Wallet firma la orden de pago. El Banco verificarÃ¡ esta firma.

    TODO equipo Wallet: exigir re-autenticaciÃ³n (PIN/biometrÃ­a/TOTP) antes de firmar.

    if body.card_id:
        card = db.get(Card, body.card_id)
        if card is None or card.user_id != user.id:
            raise HTTPException(404, "Tarjeta no encontrada")

        amount_cents=body.amount_cents,
        currency=body.currency,

/// Crea un cliente HTTP. Si `auth` es true, agrega el token JWT guardado de forma segura.
///
/// TODO equipo de seguridad: certificate pinning (comparar el hash del certificado del
/// servidor) y rechazar HTTP en producciÃ³n.
Dio buildDio(String baseUrl, SecureStore store, {bool auth = false}) {
  final dio = Dio(
    BaseOptions(
      baseUrl: baseUrl,
      connectTimeout: const Duration(seconds: 10),
      receiveTimeout: const Duration(seconds: 15),
      headers: {'Accept': 'application/json'},
    ),
  );
  if (auth) {
    dio.interceptors.add(
      InterceptorsWrapper(
        onRequest: (options, handler) async {
          final token = await store.readToken();
          if (token != null) {
            options.headers['Authorization'] = 'Bearer $token';
          }
          handler.next(options);
        },
      ),
    );
  }
  return dio;
}

/// Convierte un error de Dio en un mensaje legible para el usuario.
String errorMessage(Object error) {
  if (error is DioException) {
    final data = error.response?.data;
    if (data is Map && data['detail'] is String) {
      return data['detail'] as String;
    }
    if (error.type == DioExceptionType.connectionError ||
        error.type == DioExceptionType.connectionTimeout) {
      return 'No se pudo conectar con el servidor';
    }
  }
  return 'OcurriÃ³ un error inesperado';
}

=== mobile/lib/core/security/receipt_verifier.dart ===
import 'dart:convert';

import 'package:cryptography/cryptography.dart';

/// JSON canÃ³nico: llaves ordenadas y sin espacios. Debe producir exactamente los mismos
/// bytes que `canonical_json` del backend (backend/shared/security/signing.py).
String canonicalJson(Object? value) => jsonEncode(_sortKeys(value));

Object? _sortKeys(Object? value) {
  if (value is Map) {
    final keys = value.keys.map((k) => k.toString()).toList()..sort();
    return {for (final k in keys) k: _sortKeys(value[k])};
  }
  if (value is List) {
    return value.map(_sortKeys).toList();
  }
  return value;
}

/// Verifica firmas Ed25519 (p. ej. el recibo firmado por el Banco).
class ReceiptVerifier {
  final Ed25519 _algorithm = Ed25519();

  Future<bool> verify({
    required Map<String, dynamic> payload,
    required String signatureB64,
    required String publicKeyB64,
  }) async {
    final signature = Signature(
      base64Url.decode(signatureB64),
      publicKey: SimplePublicKey(base64Url.decode(publicKeyB64), type: KeyPairType.ed25519),
    );
    return _algorithm.verify(utf8.encode(canonicalJson(payload)), signature: signature);
  }
}
